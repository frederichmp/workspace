CREATE TABLESPACE TBS_STARTUPMONITOR DATAFILE '<diretorio>/str_monitor' SIZE 1M AUTOEXTEND ON NEXT 10M MAXSIZE 5G;

CREATE USER STARTUPMONITOR
IDENTIFIED BY STARTUPMONITOR
DEFAULT TABLESPACE TBS_STARTUPMONITOR
TEMPORARY TABLESPACE TEMP
PROFILE DEFAULT
ACCOUNT UNLOCK;

GRANT CONNECT TO STARTUPMONITOR;
GRANT RESOURCE TO STARTUPMONITOR;
ALTER USER STARTUPMONITOR DEFAULT ROLE ALL;

GRANT CREATE SESSION TO STARTUPMONITOR;	
GRANT SELECT ANY DICTIONARY TO STARTUPMONITOR;
GRANT UNLIMITED TABLESPACE TO STARTUPMONITOR;
GRANT SELECT ANY DICTIONARY TO STARTUPMONITOR;

exec dbms_network_acl_admin.create_acl(acl => 'resolve.xml',description => 'resolve acl', principal =>'STARTUPMONITOR',is_grant => true, privilege => 'resolve');
exec dbms_network_acl_admin.assign_acl(acl => 'resolve.xml', host =>'*');

select utl_inaddr.get_host_name('127.0.0.1') from dual;
