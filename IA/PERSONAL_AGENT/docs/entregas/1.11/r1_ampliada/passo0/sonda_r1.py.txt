class _CurDiag:
    def __init__(self, cfg):
        self.cfg, self.description, self._sql = cfg, None, ""

    def execute(self, sql, **binds):
        self._sql = sql
        if "COUNT(*)" in sql:
            self.description = [("COUNT(*)",)]
        elif "gv$sqlarea" in sql.lower():
            self.description = [(c,) for c in ("SQL_ID", "EXECUTIONS", "ELAPSED_SEG_TOTAL", "ELAPSED_POR_EXEC", "SQL_TEXT_CURTO")]
        elif "DBMS_XPLAN" in sql:
            self.description = [("PLAN_TABLE_OUTPUT",)]
        else:
            self.description = [(c,) for c in ("INST_ID", "SID", "serial#", "SERIAL", "USERNAME", "PROGRAM", "MACHINE", "STATUS", "SQL_ID", "EVENT")]

    def fetchone(self):
        return (3,)

    def fetchall(self):
        if "gv$sqlarea" in self._sql.lower():
            return [("abc123def4567", 10, 1.5, 0.15, "SELECT * FROM t WHERE c = 'sql-sentinela'")]
        if "DBMS_XPLAN" in self._sql:
            return [("Plan hash value: 1",), ("   1 - filter(\"C\"='literal-plano-sentinela')",)]
        return [(1, 101, 202, 202, "USUARIO_SENTINELA", "programa-sentinela", "maquina-sentinela", "ACTIVE", "abc123def4567", "ev")]

    def close(self):
        pass


class _ConDiag:
    def __init__(self, cfg):
        self.cfg = cfg

    def cursor(self):
        if self.cfg.get("erro_cursor"):
            raise self.cfg["erro_cursor"]
        return _CurDiag(self.cfg)

    def close(self):
        pass


def _instalar_diag(monkeypatch, **cfg):
    def connect(*a, **k):
        if cfg.get("erro_connect"):
            raise cfg["erro_connect"]
        return _ConDiag(cfg)
    monkeypatch.setattr(oracledb, "connect", connect)
